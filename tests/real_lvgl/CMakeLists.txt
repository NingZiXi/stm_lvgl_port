cmake_minimum_required(VERSION 3.22)
project(stm_lvgl_port_real_tests C)
enable_testing()
add_compile_options(-Wall -Wextra -UNDEBUG)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE FILEPATH "Real test configuration")
set(STM_LVGL_PORT_FETCH_LVGL OFF CACHE BOOL "Offline tests")
add_subdirectory(../.. component)
add_subdirectory(../../../stm_lcd_ili9881c ili9881c)
add_executable(test_real_render test_real_render.c)
target_link_libraries(test_real_render PRIVATE stm_lvgl_port stm_lcd_ili9881c)
foreach(_target stm_lcd stm_lcd_ili9881c stm_lvgl_port test_real_render)
    target_compile_options(${_target} PRIVATE -Werror)
endforeach()
if(NOT WIN32)
    target_link_libraries(test_real_render PRIVATE m)
endif()
add_test(NAME real_lvgl_partial_cold_start COMMAND test_real_render partial)
add_test(NAME real_lvgl_direct_cold_start COMMAND test_real_render direct)
