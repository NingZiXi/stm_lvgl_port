cmake_minimum_required(VERSION 3.22)
project(stm_lvgl_port_test LANGUAGES C CXX)
enable_testing()
add_compile_options(-Wall -Wextra -Werror -UNDEBUG)
add_library(lvgl INTERFACE)
target_include_directories(lvgl INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
add_subdirectory(.. component)
# 只在测试构建替换 libc 分配符号；生产组件没有分配器选项。
target_compile_definitions(stm_lvgl_port PRIVATE calloc=test_calloc free=test_free)
add_executable(test_component test_lvgl.c test_allocator.c)
target_link_libraries(test_component PRIVATE stm_lvgl_port)
add_library(header_consumers OBJECT headers.c headers.cpp)
target_compile_features(header_consumers PRIVATE c_std_11 cxx_std_17)
target_link_libraries(header_consumers PRIVATE stm_lvgl_port)
add_test(NAME contracts COMMAND test_component)

set(_integration_chips stm_lcd_st7789 stm_lcd_st7796 stm_lcd_ili9881c stm_lcd_touch_ft5206 stm_lcd_touch_gt9271)
set(_integration_available TRUE)
foreach(_chip IN LISTS _integration_chips)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../${_chip}/CMakeLists.txt")
        set(_integration_available FALSE)
    endif()
endforeach()
if(_integration_available)
    foreach(_chip IN LISTS _integration_chips)
        add_subdirectory("../../${_chip}" "${_chip}")
    endforeach()
    add_executable(test_chip_integration test_chip_integration.c test_allocator.c)
    target_link_libraries(test_chip_integration PRIVATE stm_lvgl_port ${_integration_chips})
    add_test(NAME five_chip_and_direct_integration COMMAND test_chip_integration)
else()
    message(STATUS "Five-chip integration requires sibling migrated chip sources")
endif()
